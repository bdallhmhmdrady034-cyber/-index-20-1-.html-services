(()=>{
 const canvas=document.getElementById("matrix");
 const ctx=canvas.getContext("2d");
 const chars="01アイウエオ<>[]{}#$%";
 let drops=[],size=16,columns=0;

 function resize(){
  canvas.width=window.innerWidth;
  canvas.height=window.innerHeight;
  columns=Math.ceil(canvas.width/size);
  drops=Array(columns).fill(1);
 }
 function draw(){
  ctx.fillStyle="rgba(5,11,9,.12)";
  ctx.fillRect(0,0,canvas.width,canvas.height);
  ctx.fillStyle="#20ff83";
  ctx.font=size+"px monospace";
  for(let i=0;i<drops.length;i++){
   let ch=chars[Math.floor(Math.random()*chars.length)];
   ctx.fillText(ch,i*size,drops[i]*size);
   if(drops[i]*size>canvas.height && Math.random()>.975) drops[i]=0;
   drops[i]++;
  }
  requestAnimationFrame(draw);
 }
 window.addEventListener("resize",resize);
 resize();
 draw();
})();
